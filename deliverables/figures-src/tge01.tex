\documentclass[border=14pt]{standalone}
\usepackage{ctex}
\usepackage{tikz}
\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm, font=\normalsize, line width=0.8pt]

% fig-ge-01 同一个量的三张脸（图内文字全中文，只保留变量符号与单位）
% 左：波形；中：旋转箭头；右：相量（复平面）
% 三栏同一个量 u = 311 sin(wt + 0 deg)

\def\colw{4.4}
\def\gap{1.3}

% ── 栏 1：波形 ──
\begin{scope}[shift={(0,0)}]
  \node[anchor=south] at ({\colw/2},3.6) {波形};
  \draw[->] (0,0) -- ({\colw},0) node[anchor=west, font=\small] {\textit{t}};
  \draw[->] (0,-1.2) -- (0,1.3) node[anchor=south, font=\small] {\textit{u}};
  \draw[domain=0:{\colw}, smooth, samples=90] plot (\x, {sin(\x/\colw*360)});
  \draw[dashed, gray, line width=0.4pt] (0,1) -- (0.5,1);
  \node[anchor=east, font=\small] at (0,1.15) {峰值 311};
  \draw[<->, line width=0.5pt] (0,-1.5) -- ({\colw},-1.5)
    node[midway, below, font=\small] {周期 T};
  \node[anchor=north, font=\small, align=center] at ({\colw/2},-2.3)
    {幅值 = 峰值 311\\周期 = T};
\end{scope}

% ── 栏 2：旋转箭头 ──
\begin{scope}[shift={({\colw+\gap},0)}]
  \node[anchor=south] at ({\colw/2},3.6) {旋转箭头};
  \draw (1.1,0) circle (1.1);
  \coordinate (O) at (1.1,0);
  \draw[->, line width=1pt] (O) -- ++(1.1,0);
  \draw[->, line width=0.6pt] (O) ++(0.55,0) arc (0:70:0.55);
  \node[font=\small] at (1.6,0.66) {\textit{w}};
  \node[anchor=north, font=\small] at (1.1,-1.2) {半径 = 311};
  \node[anchor=north, font=\small, align=center] at ({\colw/2},-2.3)
    {转速看 \textit{w}\\在 \textit{u} 轴的投影 = 瞬时值};
\end{scope}

% ── 栏 3：相量 ──
\begin{scope}[shift={({2*(\colw+\gap)},0)}]
  \node[anchor=south] at ({\colw/2},3.6) {相量};
  \draw[->] (0,0) -- ({\colw},0) node[anchor=west, font=\small] {Re};
  \draw[->] (0,0) -- (0,1.7) node[anchor=south, font=\small] {Im};
  \draw[->, line width=1pt] (0,0) -- (2.6,0);
  
  \draw[->, dashed, line width=0.7pt] (0,0) -- (1.45,1.35);
  \draw[line width=0.5pt] (0.9,0) arc (0:30:0.9);
  \node[anchor=west, font=\small] at (1.05,0.34) {初相位 $\varphi$};
  \node[anchor=north, font=\small, align=center] at ({\colw/2},-2.3)
    {模 = 311\\角 = 初相位 $\varphi$};
\end{scope}

% 栏间虚线对应
\draw[dashed, gray, line width=0.5pt] ({\colw+0.1},0) -- ({\colw+\gap-0.1},0);
\draw[dashed, gray, line width=0.5pt] ({2*\colw+\gap+0.1},0) -- ({2*\colw+2*\gap-0.1},0);

\end{tikzpicture}
\end{document}
